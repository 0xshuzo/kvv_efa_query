cmake_minimum_required(VERSION 3.16)
project(kvv_efa_query LANGUAGES CXX)
find_package(Threads REQUIRED)
include(cmake/JsonDependency.cmake)
ensure_json_dependency("${CMAKE_CURRENT_SOURCE_DIR}")
add_library(kvv_efa_query src/kvv_efa_query.cpp src/transit_refresh.cpp)
target_compile_features(kvv_efa_query PUBLIC cxx_std_17)
target_include_directories(kvv_efa_query PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(kvv_efa_query PRIVATE nlohmann_json::nlohmann_json PUBLIC Threads::Threads)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(query_checks tests/query_checks.cpp)
    target_link_libraries(query_checks PRIVATE kvv_efa_query nlohmann_json::nlohmann_json)
    add_test(NAME query_checks COMMAND query_checks)
  endif()
endif()
